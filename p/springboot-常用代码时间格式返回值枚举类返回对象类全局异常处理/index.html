一、 时间格式

默认情况下，springboot返回的时间格式是这样的：{"name":"myBean的Name","date":"2024-08-26T06:03:28.929+00:00"}

日期和时间统一设置方式如下：

1、修改配置文件配置application.yml：

spring:
  # 配置日期格式化
  jackson:
    date-format: yyyy-MM-dd HH:mm:ss  #时间戳统一转换为指定格式
    time-zone: GMT+8  # 时区修改为东8区

结果如下：{"name":"myBean的Name","date":"2024-08-26 14:06:36"}

2、@JsonFormat注解

import com.fasterxml.jackson.annotation.JsonFormat;

@JsonFormat(pattern="yyyy-MM-dd HH:mm")
private Date date;

结果如下：{"name":"myBean的Name","date":"2024-08-26 14:08"}

3、@Temporal注解：是JPA的注解，属于javax.persistence包

// 得到YYYY-MM-DD日期格式
@Temporal(TemporalType.DATE)
private Date myDate;

// 得到HH:MM:SS时间格式
@Temporal(TemporalType.TIME)
private Date myTime;

// 得到YYYY-MM-DD HH:MM:SS时间格式
@Temporal(TemporalType.TIMESTAMP)
private Date myTimestamp;

二、返回值枚举类

package com.sxdx.demo.enums;

public enum ApiResponseStatus {
    RC200("200", "成功"),
    RC201("201", "资源已创建"),
    RC204("204", "无内容"),
    RC400("400", "错误请求"),
    RC401("401", "未授权"),
    RC403("403", "禁止访问"),
    RC404("404", "未找到资源"),
    RC500("500", "内部服务器错误");

    //可补充自定义状态

    private final String code;
    private final String message;

    ApiResponseStatus(String code, String message) {
        this.code = code;
        this.message = message;
    }
    
    public String getCode() {
        return code;
    }

    public String getMessage() {
        return message;
    }

    public static ApiResponseStatus getApiResponseStatus(String code){
        for (ApiResponseStatus element:ApiResponseStatus.values()) {
            if (element.getCode().equalsIgnoreCase(code)){
                return element;
            }
        }
        return null;
    }

}

三、统一返回对象

package com.sxdx.demo.bean;

import com.sxdx.demo.enums.ApiResponseStatus;

import java.io.Serializable;

public class ResultData<T> implements Serializable {
    private static final long serialVersionUID = 1L;

    //结果状态，使用ApiResponseStatus枚举类
    private String code;
    private String message;
    private T data;
    private long timestamp;

    // 静态方法帮助创建成功响应
    public static <T> ResultData<T> success(T data) {
        ResultData resultData = new ResultData();
        resultData.setCode(ApiResponseStatus.RC200.getCode());
        resultData.setMessage("成功");
        resultData.setData(data);
        resultData.setTimestamp(System.currentTimeMillis());

        return resultData;
    }

    // 静态方法帮助创建错误响应
    public static <T> ResultData<T> error(String code) {
        ResultData resultData = new ResultData();
        resultData.setCode(code);
        resultData.setMessage(ApiResponseStatus.getApiResponseStatus(code).getMessage());
        resultData.setData(null);
        resultData.setTimestamp(System.currentTimeMillis());

        return resultData;
    }

    // 静态方法帮助创建错误响应
    public static <T> ResultData<T> error(String code,T data) {
        ResultData resultData = new ResultData();
        resultData.setCode(code);
        resultData.setMessage(ApiResponseStatus.getApiResponseStatus(code).getMessage());
        resultData.setData(data);
        resultData.setTimestamp(System.currentTimeMillis());

        return resultData;
    }
    // 静态方法帮助创建错误响应
    public static <T> ResultData<T> error(String code,String message,T data) {
        ResultData resultData = new ResultData();
        resultData.setCode(code);
        resultData.setMessage(message);
        resultData.setData(data);
        resultData.setTimestamp(System.currentTimeMillis());

        return resultData;
    }

    public ResultData() {
        this.timestamp = System.currentTimeMillis();
    }

    public String getCode() {
        return code;
    }

    public void setCode(String code) {
        this.code = code;
    }

    public String getMessage() {
        return message;
    }

    public void setMessage(String message) {
        this.message = message;
    }

    public T getData() {
        return data;
    }

    public void setData(T data) {
        this.data = data;
    }

    public long getTimestamp() {
        return timestamp;
    }

    public void setTimestamp(long timestamp) {
        this.timestamp = timestamp;
    }
}

 @RequestMapping("/getdate")
 public ResultData<MyBean> getDate(){
 	MyBean myBean = new MyBean("myBean的Name");
 	myBean.setDate(new Date());

	return ResultData.success(myBean);

四、全局异常处理

4.1 定义自定义异常类

package com.sxdx.demo.exception;

// 自定义异常类
public class CustomNotFoundException extends RuntimeException {
    public CustomNotFoundException(String message) {
        super(message);
    }
}

4.2 全局异常类

package com.sxdx.demo.exception;

import com.sxdx.demo.bean.ResultData;
import com.sxdx.demo.enums.ApiResponseStatus;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ControllerAdvice;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.ResponseStatus;

@ControllerAdvice
public class GlobalExceptionHandler {
    // 处理自定义异常
    @ExceptionHandler(CustomNotFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public ResponseEntity<ResultData> handleCustomNotFoundException(CustomNotFoundException ex) {
        ResultData resultData = ResultData.error(ApiResponseStatus.RC404.getCode(),ex.getMessage());
        return new ResponseEntity<>(resultData, HttpStatus.NOT_FOUND);
    }

    // 处理所有其他运行时异常
    @ExceptionHandler(RuntimeException.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public ResponseEntity<ResultData> handleNullPointerException(RuntimeException ex) {
        ResultData resultData = ResultData.error(ApiResponseStatus.RC500.getCode(),ex.getMessage(),ex.toString());
        return new ResponseEntity<>(resultData, HttpStatus.INTERNAL_SERVER_ERROR);
    }


    // 处理所有其他未处理的异常
    @ExceptionHandler(Exception.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public ResponseEntity<ResultData> handleGenericException(Exception ex) {
        ResultData resultData = ResultData.error(ApiResponseStatus.RC999.getCode(),ex.getMessage(),ex.toString());
        return new ResponseEntity<>(resultData, HttpStatus.INTERNAL_SERVER_ERROR);
    }
}

注意：@ResponseStatus 和 ResponseEntity同时存在时，ResponseEntity的HttpStatus参数优先级更高。
